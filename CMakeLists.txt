cmake_minimum_required(VERSION 3.16)
project(LensSimulator VERSION 0.2.0 LANGUAGES CXX)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(LENS_BUILD_QT "Build the optional Qt desktop viewer" ON)
option(LENS_BUILD_EXAMPLES "Build the exploratory optics CLI" ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()

add_library(optics_core INTERFACE)
target_include_directories(optics_core INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/core)
target_compile_features(optics_core INTERFACE cxx_std_17)

add_library(lens_model STATIC lens.cpp core/lensanalysis.cpp
  raytracer/ray.cpp raytracer/plane.cpp raytracer/shape.cpp)
target_include_directories(lens_model PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_include_directories(lens_model SYSTEM PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}/glm ${CMAKE_CURRENT_SOURCE_DIR}/raytracer)
target_link_libraries(lens_model PUBLIC optics_core)
if(MSVC)
  target_compile_options(lens_model PRIVATE /utf-8 /W3)
endif()

if(LENS_BUILD_EXAMPLES)
  add_executable(optics_workbench apps/workbench.cpp)
  target_link_libraries(optics_workbench PRIVATE optics_core)
endif()

if(LENS_BUILD_QT)
  find_package(Qt6 QUIET COMPONENTS Widgets Svg)
  if(Qt6_FOUND)
    set(LENS_QT Qt6)
  else()
    find_package(Qt5 5.9 REQUIRED COMPONENTS Widgets Svg)
    set(LENS_QT Qt5)
  endif()
  add_library(lens_viewer STATIC mainwindow.cpp mainwindow.h
    lensprofileview.cpp lensprofileview.h)
  set_target_properties(lens_viewer PROPERTIES AUTOMOC ON)
  target_link_libraries(lens_viewer PUBLIC lens_model ${LENS_QT}::Widgets ${LENS_QT}::Svg)
  add_executable(QtLensSimulator WIN32 MACOSX_BUNDLE main.cpp)
  target_link_libraries(QtLensSimulator PRIVATE lens_viewer)
  add_custom_command(TARGET QtLensSimulator POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      ${CMAKE_CURRENT_SOURCE_DIR}/LensPrescription $<TARGET_FILE_DIR:QtLensSimulator>/LensPrescription)
  if(WIN32)
    get_target_property(qmake_path ${LENS_QT}::qmake IMPORTED_LOCATION)
    get_filename_component(qt_bin "${qmake_path}" DIRECTORY)
    find_program(LENS_WINDEPLOYQT windeployqt HINTS "${qt_bin}")
    if(LENS_WINDEPLOYQT)
      add_custom_target(deploy_qt
        COMMAND "${LENS_WINDEPLOYQT}" --no-translations --no-compiler-runtime
          $<IF:$<CONFIG:Debug>,--debug,--release> $<TARGET_FILE:QtLensSimulator>
        DEPENDS QtLensSimulator COMMENT "Deploying Qt runtime beside the viewer")
    endif()
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(optics_tests tests/physics_tests.cpp)
  target_link_libraries(optics_tests PRIVATE optics_core)
  add_test(NAME physics COMMAND optics_tests)
  add_executable(lens_tests tests/lens_tests.cpp)
  target_link_libraries(lens_tests PRIVATE lens_model)
  add_test(NAME lens_integration COMMAND lens_tests ${CMAKE_CURRENT_SOURCE_DIR}/LensPrescription)
  if(LENS_BUILD_QT)
    add_executable(viewer_tests tests/viewer_tests.cpp)
    target_link_libraries(viewer_tests PRIVATE lens_viewer)
    add_test(NAME viewer_integration COMMAND viewer_tests ${CMAKE_CURRENT_SOURCE_DIR}/LensPrescription/lens0.txt)
    set_tests_properties(viewer_integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    if(WIN32)
      set_property(TEST viewer_integration APPEND PROPERTY ENVIRONMENT
        "QT_QPA_PLATFORM_PLUGIN_PATH=${qt_bin}/../plugins/platforms")
    endif()
  endif()
endif()
